# Barkley's AppArmor profile (app slug sniff_test). The Supervisor loads it when the app
# is installed or updated (never on Rebuild) and rewrites the profile name below to the
# app's id (local_sniff_test here; <repository hash>_sniff_test from a store). It takes
# the security rating from 7 to 8, but the point is the scanner: it parses hostile code
# with Semgrep, GuardDog, osv-scanner and uv, and a hole in one of those parsers must not
# be able to do more than this file allows. Barkley, s6 and bashio run under the main
# profile; on a kernel without Landlock (HA OS) every scanner tool runs in the child
# profile `tools` at the end of this file.
#
# What it allows, and why:
# - s6-overlay and bashio: the paths from HA's own example profile. HA's example also
#   grants `file,` (every file, every permission) at the top, which is why its `ix` lines
#   never needed `r`; this profile has no `file,`, so the scripts and binaries that are
#   read as well as run (execline scripts read themselves, uv reads the shell and Python
#   it finds on PATH, osv-scanner reads its own binary) carry `r` explicitly.
# - Run only what the image ships: the image's Python 3.13, the scanner's venv, the
#   tools' venvs, uv and osv-scanner. Nothing under /tmp, /data or HOME is executable.
# - Write only under /data (the store, options.json, the caches; HOME is moved to
#   /data/cache/home by the run script so Semgrep's settings and log and uv's credential
#   store land there; GuardDog's typosquat lists refresh into /data/cache/guarddog), /tmp
#   (one workspace per scan), and s6-overlay's own folders under /run and /etc (HA's
#   example profile's rules, which s6 needs to start the service). /opt and /usr are
#   never written.
# - The scanner tools get much less (child profile `tools`, used where the kernel has no
#   Landlock; with Landlock, Barkley's own Landlock sandbox confines them instead, see
#   src/sniff_test/sandbox.py). s6's with-contenv keeps the container environment,
#   SUPERVISOR_TOKEN included, as files under /run/s6/container_environment, and the
#   main profile must let s6 read them; /data holds options.json, the GitHub token file
#   and the scan store. A hole in a tool's parser must reach none of them, so `tools`
#   has no /run, no /etc writes, and in /data only the caches. Barkley starts each tool
#   through /usr/lib/barkley/tool, which moves into `tools` on exec. A move into another
#   profile is refused under no-new-privileges (HA OS and the Supervisor don't set it);
#   Barkley then finds the tools unsandboxed and doesn't scan (src/sniff_test/sandbox.py).
# - Network: TCP to GitHub, PyPI, HA's wheel index, OSV, the npm registry and its
#   download counts, jsDelivr's hash lookup and the CVE records at cveawg.mitre.org
#   (0.14.1), the Supervisor's API and Core proxy,
#   and the ingress panel on 8099; UDP and netlink for DNS. No raw sockets. The rules are
#   by socket type, not by host: the hosts are limited in Barkley's code
#   (fetch/http.py ALLOWED_HOSTS).
# - No capabilities, no mount, no ptrace.
#
# Grounded in an strace of doctor, the four fixtures and two real scans (an
# integration and a card) in the arm64 image on 2026-11-11, then one logging-only run
# of 0.7.7 on the user's HA OS box (Diagnostics, 16 re-sniffs, two sniffs from search, a
# test notification, Explain in context): 9,304 would-be denials, all of them reads, in
# three groups, each allowed below with its cause. 0.7.8 enforces. HA OS 18.4 runs
# AppArmor 3.1.7: no abi line, nothing newer than 3.0 syntax.
#include <tunables/global>

profile sniff_test flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  # DNS and the sockets for it: inet/inet6 stream and dgram, netlink raw; /etc/hosts,
  # resolv.conf, nsswitch.conf, gai.conf; libnss.
  #include <abstractions/nameservice>

  # Signals: s6 supervises its service, the scanner kills a tool that overran its
  # timeout, Go's runtime (osv-scanner) signals its own threads; `docker stop` sends
  # TERM from the host.
  signal (send, receive) peer=@{profile_name},
  signal (receive) peer=unconfined,
  # the scanner kills a tool in `tools` that overran its timeout
  signal (send) peer=@{profile_name}//tools,

  # Unix sockets: only the tools' own IPC (Python's multiprocessing in Semgrep); base
  # already limits unix peers to this profile.
  network unix stream,
  network unix dgram,

  # --- s6-overlay and bashio (HA's example profile's paths, with `r` where the box's
  # logging-only run showed a read: /init and s6's stage scripts are read by s6 itself,
  # execline scripts such as printcontenv re-open themselves, s6-rc-compile reads the
  # service definitions under /package, and the services.d / fix-attrs.d folders are
  # listed) ---
  /init rix,
  /bin/** rix,
  /usr/bin/** rix,
  /run/{s6,s6-rc*,service}/** rix,
  /package/{,**} rix,
  /command/{,**} rix,
  /etc/services.d/{,**} rwix,
  /etc/cont-init.d/{,**} rwix,
  /etc/cont-finish.d/{,**} rwix,
  /etc/fix-attrs.d/{,**} r,
  /etc/s6-overlay/{,**} rwix,
  /run/{,**} rwk,
  /dev/tty rw,
  /usr/lib/bashio/** ix,
  /tmp/** rwk,
  /data/** rw,

  # --- Barkley ---
  # The image's Python 3.13, the scanner's venv and the tools' venvs: read, map their
  # native modules, run their launchers. Semgrep's engine (semgrep-core) is a native
  # binary inside its package, and it maps 26 bundled libraries named like
  # libcrypto.so.3 and libtree-sitter.so.0.26 (pygit2 and yara-python bundle theirs the
  # same way), so the map rules take any `.so.*` suffix.
  /usr/lib/python3.13/{,**} r,
  /usr/lib/python3.13/lib-dynload/*.so{,.*} mr,
  /usr/lib/python3/{,**} r,
  /usr/local/lib/python3.13/{,**} r,
  # (0.14.0) Barkley's own Python (HA's version), installed by uv at build time.
  /opt/python/{,**} r,
  /opt/python/*/bin/* ix,
  /opt/python/barkley-python ix,
  /opt/python/**.so{,.*} mr,
  /opt/sniff-test/{,**} r,
  /opt/sniff-test/bin/* ix,
  /opt/sniff-test/**.so{,.*} mr,
  /opt/tools/{,**} r,
  /opt/tools/*/bin/* ix,
  /opt/tools/**.so{,.*} mr,
  /opt/tools/semgrep/lib/python3*/site-packages/semgrep/bin/* ix,
  # uv and osv-scanner, plus the venv launchers linked here. osv-scanner reads its own
  # binary at start (Go's build info), so `r` as well as exec. (With Landlock the tools
  # run from here, inside Barkley's Landlock sandbox.)
  /usr/local/bin/* rix,
  # The way into `tools` (without Landlock): a two-line script that execs the tool.
  /usr/lib/barkley/tool rcx -> tools,
  # uv looks for Python interpreters by listing every directory on PATH, then reads the
  # ones it finds (/usr/bin/python3.13, and the shell), so the PATH directories are
  # listable. uv also learns the C library's version by running the dynamic loader
  # (`ld.so --version`).
  /{usr/,}{s,}bin/ r,
  /usr/local/{s,}bin/ r,
  /{usr/,}lib{,64}/ld-linux-*.so* ix,
  /{usr/,}lib/@{multiarch}/ld-linux-*.so* ix,

  # The store (SQLite needs locks), options.json, the caches (HA's constraints, uv,
  # GuardDog's lists), HOME; and one temp workspace per scan. Never executable.
  /data/{,**} rwk,
  /tmp/{,**} rwk,
  # osv-scanner 2.7.1 reaches a lockfile by opening `/` as a directory and walking down
  # from it (`/`, `tmp`, the scan's folder, the file), where 2.7.0 opened the path
  # directly (strace, 2026-11-14). Without this it reads no lockfile at all ("could not
  # determine extractor suitable to this file", exit 127; seen on a box running 0.10.3).
  # Listing `/` shows the names of the top-level folders, nothing in them.
  / r,

  # What the tools read: TLS roots (urllib, curl, uv, osv-scanner), the OS id (uv),
  # terminfo (bashio's shell), MIME types (urllib).
  /etc/ssl/{,**} r,
  /usr/lib/ssl/{,**} r,
  /usr/share/ca-certificates/{,**} r,
  /etc/os-release r,
  /usr/lib/os-release r,
  /etc/mime.types r,
  /etc/inputrc r,
  /etc/terminfo/{,**} r,
  /usr/share/terminfo/{,**} r,
  # Facts the panel reads: free memory (the memory guard) is in base; the profile the
  # process runs under (diagnostics). Python's subprocess lists its own fds; uv and Go
  # read their cgroup limits and their own exe path.
  @{PROC}/@{pid}/attr/{,apparmor/}current r,
  @{PROC}/@{pid}/fd/ r,
  @{PROC}/@{pid}/cgroup r,
  @{PROC}/@{pid}/mountinfo r,
  @{PROC}/@{pid}/exe r,
  @{sys}/fs/cgroup/{,**} r,
  @{sys}/devices/system/cpu/{,**} r,
  @{sys}/kernel/mm/transparent_hugepage/{,*} r,

  # --- the scanner tools without Landlock (Semgrep, GuardDog, osv-scanner, uv) ---
  # Entered through /usr/lib/barkley/tool. Everything a scan needs, nothing that holds a
  # credential. The deny lines repeat what the missing allow rules already mean, so a
  # later allow rule can't open them by accident. (Child profiles are indented: the
  # Supervisor renames only the line that starts with `profile`.)
  profile tools flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/nameservice>

    deny /run/s6/container_environment/{,**} rwkl,
    deny /data/options.json rwkl,
    deny /data/github-token rwkl,
    deny /data/sniff-test.db* rwkl,

    # Barkley stops a tool that overran; a tool signals its own processes (Go's runtime).
    signal (receive),
    signal (send) peer=@{profile_name},
    network unix stream,
    network unix dgram,

    # the launcher (the shell reads it) and the tools: the same rules as the main profile
    /usr/lib/barkley/tool r,
    /bin/** rix,
    /usr/bin/** rix,
    /usr/lib/python3.13/{,**} r,
    /usr/lib/python3.13/lib-dynload/*.so{,.*} mr,
    /usr/lib/python3/{,**} r,
    /usr/local/lib/python3.13/{,**} r,
    /opt/tools/{,**} r,
    /opt/tools/*/bin/* ix,
    /opt/tools/**.so{,.*} mr,
    /opt/tools/semgrep/lib/python3*/site-packages/semgrep/bin/* ix,
    /usr/local/bin/* rix,
    /{usr/,}{s,}bin/ r,
    /usr/local/{s,}bin/ r,
    /{usr/,}lib{,64}/ld-linux-*.so* ix,
    /{usr/,}lib/@{multiarch}/ld-linux-*.so* ix,
    # Barkley's own Semgrep rules (read only; the rest of Barkley stays out of reach)
    /opt/sniff-test/lib/python3*/site-packages/sniff_test/{rules,rules_const}/{,**} r,

    # the caches (uv's packages, HA's constraints, GuardDog's lists) and the workspaces;
    # HOME and TMPDIR are moved inside the scan's workspace. Never executable.
    /data/cache/{,**} rwk,
    /tmp/{,**} rwk,
    / r,

    /etc/ssl/{,**} r,
    /usr/lib/ssl/{,**} r,
    /usr/share/ca-certificates/{,**} r,
    /etc/os-release r,
    /usr/lib/os-release r,
    /etc/mime.types r,
    /etc/inputrc r,
    /etc/terminfo/{,**} r,
    /usr/share/terminfo/{,**} r,
    /dev/tty rw,
    @{PROC}/@{pid}/attr/{,apparmor/}current r,
    @{PROC}/@{pid}/fd/ r,
    @{PROC}/@{pid}/cgroup r,
    @{PROC}/@{pid}/mountinfo r,
    @{PROC}/@{pid}/exe r,
    @{sys}/fs/cgroup/{,**} r,
    @{sys}/devices/system/cpu/{,**} r,
    @{sys}/kernel/mm/transparent_hugepage/{,*} r,
  }
}
